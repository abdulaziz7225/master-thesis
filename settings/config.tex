% Render \cite{a,b,c} as [1, 2, 3] (single bracket pair) rather than IEEE's
% default [1], [2], [3]. Wraps the whole iteration in one \mkbibbrackets and
% separates entries with a comma + space.
\DeclareCiteCommand{\cite}[\mkbibbrackets]
  {\usebibmacro{prenote}}
  {\usebibmacro{citeindex}%
   \printtext[bibhyperref]{\printfield{labelnumber}}}
  {\addcomma\space}
  {\usebibmacro{postnote}}
